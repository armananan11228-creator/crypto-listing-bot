// ═══════════ موتور تحلیل اسپات (ترکیب هر چهار اندیکاتور قبلی) ═════════════
// خروجی: [امتیاز، قیمت، هدف۱، هدف۲، حدضرر، RSI، رشد حجم٪، ATR٪]
f_spot() =>
    // ── پایه‌ها ──
    float e20   = ta.ema(close, 20)
    float e50   = ta.ema(close, 50)
    float e200  = ta.ema(close, 200)
    float rsiV  = ta.rsi(close, 14)
    float atrV  = ta.atr(14)
    [mLine, mSig, macdH] = ta.macd(close, 12, 26, 9)
    [diP, diM, adxV]  = ta.dmi(14, 14)
    float vMA   = ta.sma(volume, 20)
    float vRec  = ta.sma(volume, 10)
    float vPri  = vRec[10]
    float hi20  = ta.highest(high, 20)
    float lo20  = ta.lowest(low, 20)
    float lo10  = ta.lowest(low, 10)
    float hh10  = ta.highest(high, 10)[1]

    float macdGap = mLine - mSig
    float dmiGap  = diP - diM

    // ── ۱) روند (اندیکاتور دوم) ──
    float trendS = 0.0
    trendS += (close > e50  ? 0.4 : 0.0)
    trendS += (e50   > e200 ? 0.4 : 0.0)
    trendS += (close > e200 ? 0.2 : 0.0)

    // ── ۲) مومنتوم ──
    float momoS = 0.0
    momoS += (macdH > 0 ? 0.55 : 0.0)
    momoS += (macdH > nz(macdH[1]) ? 0.30 : 0.0)
    momoS += (macdGap > 0 and dmiGap > 0 ? 0.15 : 0.0)

    // ── ۳) RSI در ناحیه سالم (نه کف مرده، نه اشباع) ──
    float rsiS = rsiV >= 45 and rsiV <= 63 ? 1.0 :
                 rsiV > 63  and rsiV <= 70 ? 0.55 :
                 rsiV >= 38 and rsiV < 45  ? 0.6 : 0.1

    // ── ۴) حجم افزایشی و تازه (اندیکاتور سوم و چهارم) ──
    float vChg = vPri > 0 ? (vRec - vPri) / vPri * 100 : 0.0
    float vRat = vMA  > 0 ? volume / vMA : 1.0
    float volS = vChg >= 10 and vChg <= 110 ? clamp(vChg / 60, 0.35, 1.0) :
                 vChg > 110 ? 0.35 : vChg > 0 ? 0.25 : 0.0
    volS := vRat > 3.0 ? volS * 0.5 : volS   // حجم منفجرشده = دیر شده

    // ── ۵) ناحیه ارزش / پولبک سالم (اردربلاک) ──
    float rng  = hi20 - lo20
    float pos  = rng > 0 ? (close - lo20) / rng * 100 : 50.0
    float pullS = pos >= 30 and pos <= 72 ? 1.0 :
                  pos > 72 and pos <= 88  ? 0.5 :
                  pos < 30 and pos >= 15  ? 0.55 : 0.15

    // ── ۶) قدرت روند ──
    float adxS = clamp((nz(adxV) - 14) / 14, 0.0, 1.0)

    // ── ۷) ساختار: کف‌های بالاتر یا شکست سقف ──
    float strS = 0.0
    strS += (lo10 > nz(lo20) ? 0.5 : 0.0)
    strS += (close > nz(hh10) ? 0.5 : 0.0)

    // ── ۸) نوسان منطقی (نه مرده، نه دیوانه) ──
    float atrP = close > 0 ? atrV / close * 100 : 0.0
    float volaS = atrP >= 0.4 and atrP <= 5.0 ? 1.0 : atrP < 0.4 ? 0.3 : 0.25

    // ── امتیاز نهایی وزنی (۰ تا ۱۰۰) ──
    float raw = trendS * 2.0 + momoS * 1.5 + rsiS * 1.2 + volS * 1.5 +
                pullS  * 1.2 + adxS  * 1.0 + strS * 0.8 + volaS * 0.8
    float sc  = clamp(raw * 10.0, 0.0, 100.0)

    // ── برنامه خروج ──
    float tp1 = close + atrV * tp1Mult
    float tp2 = close + atrV * tp2Mult
    float sl  = math.min(close - atrV * slMult, lo10 * 0.998)

    [sc, close, tp1, tp2, sl, rsiV, vChg, atrP]

             bgcolor = color.new(#2962ff, 80))

             textalign = text.align_left)

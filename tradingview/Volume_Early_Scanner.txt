clamp(float x, float lo, float hi) => math.max(lo, math.min(hi, x))
shortSym(string s) =>
    int    p = str.pos(s, ":")

// ═══════════════ موتور تحلیل «حجم در ابتدای راه» ═════════════════════════
// خروجی: [امتیاز، رشد حجم٪، نسبت کندل فعلی، RSI، رشد قیمت٪، تازگی(کندل)]
f_early() =>
    float vMA    = ta.sma(volume, volLen)
    float vLong  = ta.sma(volume, longLen)
    float recent = ta.sma(volume, cmpLen)
    float prior  = recent[cmpLen]
    float chg    = nz(prior) > 0 ? (recent - prior) / prior * 100 : 0.0
    float vr     = vMA > 0 ? volume / vMA : 0.0
    float baseQ  = vLong > 0 ? nz(prior, vLong) / vLong : 1.0      // <1 یعنی قبلش آرام بوده
    float r      = ta.rsi(close, 14)
    float a      = math.max(ta.atr(14), syminfo.mintick)
    float e      = ta.ema(close, 20)
    float extn   = math.abs(close - e) / a
    float pchg   = close[cmpLen] > 0 ? (close - close[cmpLen]) / close[cmpLen] * 100 : 0.0
    float fresh  = nz(ta.barssince(ta.crossover(chg, minChg)), 999)

    // فشار خرید (دلتای حجم)
    float rng  = math.max(high - low, syminfo.mintick)
    float cv   = ta.cum(((close - open) / rng) * volume)
    float cs   = ta.linreg(cv, 20, 0) - ta.linreg(cv, 20, 1)
    float bp   = clamp(cs / math.max(ta.stdev(cv - cv[1], 50), 1e-10) / 2, -1, 1)

    // ── اجزای امتیاز (هر کدام ۰ تا ۱) ──
    float sUp   = chg < minChg ? 0.0 : chg <= maxChg ? 1.0 : clamp(1 - (chg - maxChg) / maxChg, 0, 1)
    float sCool = vr <= hotRatio ? 1.0 : clamp(1 - (vr - hotRatio) / hotRatio, 0, 1)
    float sBase = baseQ <= 1.0 ? 1.0 : clamp(1 - (baseQ - 1) / 0.8, 0, 1)
    float sFrsh = fresh <= freshBars ? 1.0 : clamp(1 - (fresh - freshBars) / (freshBars * 3.0), 0, 1)
    float sRoom = r <= maxRsi ? 1.0 : clamp(1 - (r - maxRsi) / 20.0, 0, 1)
    float sExt  = extn <= 2.0 ? 1.0 : clamp(1 - (extn - 2.0) / 2.0, 0, 1)
    float sRun  = math.abs(pchg) <= maxRun ? 1.0 : clamp(1 - (math.abs(pchg) - maxRun) / maxRun, 0, 1)
    float sBuy  = clamp(0.5 + bp / 2.0, 0, 1)

    float sc = (sUp * 2.0 + sFrsh * 1.6 + sCool * 1.4 + sBase * 1.2 + sRoom * 1.0 + sRun * 1.0 + sExt * 0.6 + sBuy * 0.6) /
         (2.0 + 1.6 + 1.4 + 1.2 + 1.0 + 1.0 + 0.6 + 0.6) * 100
    [sc, chg, vr, r, pchg, fresh]

// وضعیت بر اساس مقادیر
stateOf(float sc, float chg, float vr, float r, float fresh) =>
    string st = "—"
    if chg <= -minChg
        st := "💤 حجم کاهشی"
    else if vr > hotRatio * 1.4 or chg > maxChg * 1.5 or r > 75
        st := "⚠️ داغ / دیر"
    else if sc >= minScore and chg >= minChg and fresh <= freshBars
        st := "🌱 تازه شروع شده"
    else if sc >= minScore * 0.8 and chg >= minChg
        st := "✅ در حال رشد"
    else if chg >= minChg
        st := "… ضعیف"
    st
